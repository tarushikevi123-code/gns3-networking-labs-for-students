#!/usr/bin/env bash
# ISP-Router - External IPv6 network simulation
# Lab 01: IPv6 Static Routing with GNS3
# Run as root inside the ISP-Router Linux node.
set -e

# Bring the R3-facing interface up
ip link set eth0 up

# Configure the external-link address and test destination
ip -6 addr replace 2001:db8:30:1::2/64 dev eth0
ip -6 addr replace 2001:db8:30:ffff::1/128 dev lo

# Enable IPv6 forwarding
sysctl -w net.ipv6.conf.all.forwarding=1

# Return route to the internal network through R3
ip -6 route replace 2001:db8:10::/48 via 2001:db8:30:1::1 dev eth0

# Verification
ip -br -6 addr
ip -6 route
